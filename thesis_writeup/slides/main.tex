\begin{frame}{The design gap}{Why another truss-optimisation thesis?}
  \begin{columns}[T,onlytextwidth]
    \column{0.57\linewidth}
      \begin{itemize}\setlength\itemsep{0.5em}
        \item India builds thousands of industrial sheds, transmission towers, and highway overpasses each year under IS~800:2007.
        \item Classical trial-and-check design typically carries \textbf{10--30\,\% surplus steel} vs.\ the theoretical feasible optimum.
        \item Steel embodied CO\textsubscript{2} $\approx \num{1.9}$~kg/kg; construction $\sim\!\num{11}\,\%$ of global emissions.
        \item A \textbf{15\,\%} cut over India's tens of thousands of small-to-medium truss structures is sector-level impact.
        \item Existing optimisation tools exist but sit on the researcher's bench --- a full GA run is ${\sim}40$~min/benchmark, too slow for iterative design.
      \end{itemize}

\begin{frame}{Problem formulation}{Weight minimisation under codal constraints}
  \vspace{0.3em}
  \begin{equation*}
    \min_{A\in\mathcal{A}}\; W(A) = \sum_{i=1}^{N}\rho\,L_i\,A_i
    \quad\text{s.t.}\quad
    \begin{cases}
      |\sigma_i(A)| \le \sigma_{\text{all},i}(A) & \text{IS~800 \S\S 6.2,\,6.3,\,7.1}\\[0.15em]
      |\delta_j(A)| \le \delta_{\max} & \text{IS~800 \S 5.6.1}\\[0.15em]
      \lambda_i(A) \le 180 & \text{IS~800 \S 3.8}\\[0.15em]
      A_i \in [A_{\min},\,A_{\max}]
    \end{cases}
  \end{equation*}
  \vspace{0.2em}
  \small
  \textbf{Four benchmarks:}\quad
  \pillblue{10-bar planar, 10 DoF} \
  \pillblue{25-bar spatial, 8 groups} \
  \pillblue{72-bar tower, 16 groups} \
  \pillblue{200-bar stepped tower, 29 groups}

  \vspace{0.7em}
  \textbf{Hard constraints only:} \ no soft penalties. Infeasible designs are always dominated by any feasible design in the tournament (pymoo).
\end{frame}

\begin{frame}{Layer 2 \& 3 \ \textendash\ FEM kernel and IS~800 compliance}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\linewidth}
      \textbf{FEM} \ (Appendix A, 10 pp)
      \begin{itemize}\small
        \item Global stiffness assembly $K=\sum_e B_e^\top k_e B_e$.
        \item Partition method enforces supports.
        \item Solve $K_{ff}u_f = F_f$ via dense \texttt{numpy.linalg.solve}.
        \item Axial stress $\sigma_e = (E_e/L_e)\,[-c\ -s\ c\ s]\,u_e$.
      \end{itemize}
      \centering
      \vspace{0.2em}
      \includegraphics[width=0.86\linewidth]{fig_3_1_truss_element.png}
    \column{0.48\linewidth}
      \textbf{IS~800:2007} \ (Appendix B, 8 pp)
      \begin{itemize}\small
        \item Tension yield $T_{dg}=A_g f_y/\gamma_{m0}$ \ \S 6.2
        \item Tension rupture $T_{dn}=0.9 A_n f_u/\gamma_{m1}$ \ \S 6.3
        \item Compression buckling via Perry-Robertson, curve~$a$ \ \S 7.1
        \item Slenderness $\lambda\le 180$ \ \S 3.8
        \item Serviceability $\delta\le L/325$ \ \S 5.6.1
      \end{itemize}
      \centering

\end{frame}

\begin{frame}{Layer 5 \ \textendash\ Neural surrogate}{MLP drops in for FEM at $132\times$ wall-clock speedup}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\linewidth}
      \begin{itemize}\small
        \item \textbf{Architecture:} MLP $[n_A] \to 256 \to 128 \to 64 \to 3$, ReLU, dropout $p{=}0.2$.
        \item \textbf{Training:} 10\,000 Latin-hypercube samples, Adam lr${=}10^{-3}$, 200 epochs.
        \item \textbf{Outputs:} weight, max-stress, max-displacement (multi-head).
        \item \textbf{Use:} drop-in FEM replacement in pymoo inner loop. \textbf{Hybrid mode:} surrogate screen $\to$ FEM fallback when feasibility is borderline.
      \end{itemize}
    \column{0.45\linewidth}
      \includegraphics[width=\linewidth]{fig_3_5_surrogate_arch.png}
  \end{columns}
\end{frame}

\begin{frame}{Layers 6 \& 7 \ \textendash\ RL agent and design-agent warm-start}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\linewidth}
      \textbf{PPO MDP}
      \begin{itemize}\footnotesize
        \item State: current $A\in\mathbb{R}^{n_A}$, $\sigma_{\max}$, $\delta_{\max}$.
        \item Action: multiplicative adjustment $\in[0.5,2.0]$.
        \item Reward: $-W - \lambda\cdot\text{infeas}$.
      \end{itemize}
      \centering\includegraphics[width=0.95\linewidth]{fig_3_6_rl_mdp.png}
    \column{0.49\linewidth}
      \textbf{design agent}
      \begin{itemize}\footnotesize
        \item Structured prompt $\to$ Claude returns $k{=}8$ candidate designs + reasoning.
        \item Injected as first-population seeds.
        \item All responses cached under \texttt{results/llm\_cache/} for reproducibility.
      \end{itemize}
      \centering\includegraphics[width=0.95\linewidth]{fig_3_7_llm_pipeline.png}
  \end{columns}
\end{frame}

\sectioncover{6}{Conclusion}{Limitations, future work, and a concrete ask.}

\begin{frame}{What I am \emph{not} claiming}{Limitations stated honestly}
  \small
  \begin{itemize}
    \item FEM kernel is \textbf{linear elastic}, small-displacement. No geometric/material non-linearity, no post-buckling, no dynamics.
    \item Optimisation is \textbf{sizing-only}, not topology. Bar connectivity is fixed by the benchmark definition.
    \item NSGA-II 25-bar and 72-bar runs used \textbf{3 seeds each} --- statistical power for O2 is weaker than for O1.
    \item PPO is trained and evaluated on one problem instance; \textbf{no transfer learning} across benchmarks.
    \item IS~800 compliance module uses a \textbf{single buckling curve} (curve~$a$) for all cross-section types.
    \item agent is \textbf{not fine-tuned} --- public Claude API at temperature 0.3, zero-shot.
  \end{itemize}
  \vspace{0.4em}
  \color{inkmuted}\footnotesize None of these are load-bearing for C1--C5; but they do set the boundary of claims I will defend.
\end{frame}
